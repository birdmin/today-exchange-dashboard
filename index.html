// KST 기준 YYYY-MM-DD 문자열 생성 함수 function getKSTDateString(dateObj) { return new Intl.DateTimeFormat('sv-SE', { timeZone: 'Asia/Seoul' }).format(dateObj); } async function fetchExchangeRate() { const fetchTime = new Date(); try { // 1. 오늘 날짜 및 어제 날짜(KST 기준) 자동 계산 const today = new Date(); const yesterday = new Date(today); yesterday.setDate(today.getDate() - 1); const todayStr = getKSTDateString(today); const yesterdayStr = getKSTDateString(yesterday); // 2. 최신(오늘) 환율 수집 const responseToday = await fetch('https://open.er-api.com/v6/latest/USD'); const dataToday = await responseToday.json(); if (dataToday.result === 'success') { state.status = 'fresh'; state.error_code = 'none'; state.last_known_rate = dataToday.rates.KRW; state.last_known_source_time = new Date(dataToday.time_last_update_unix * 1000); // 3. 기록에 어제 데이터가 없는 경우 (시크릿 모드 / 첫 접속) // 어제 날짜 API를 직접 호출하여 어제 환율을 동적으로 가져옴 const hasYesterdayRecord = state.daily_records.some(r => r.record_date === yesterdayStr); if (!hasYesterdayRecord) { try { // open.er-api의 과거 날짜 환율 호출 (YYYY/MM/DD) const [year, month, day] = yesterdayStr.split('-'); const resYesterday = await fetch(`https://open.er-api.com/v6/history/USD/${year}/${month}/${day}`); const dataYesterday = await resYesterday.json(); if (dataYesterday.result === 'success' && dataYesterday.rates && dataYesterday.rates.KRW) { state.daily_records.push({ record_date: yesterdayStr, normalized_value: dataYesterday.rates.KRW }); } } catch (e) { console.warn("어제 환율 API 수집 실패, 오늘 기록만 표시됩니다."); } } // 4. 오늘 환율 기록 업데이트 const existingIdx = state.daily_records.findIndex(r => r.record_date === todayStr); if (existingIdx >= 0) { state.daily_records[existingIdx].normalized_value = state.last_known_rate; } else { state.daily_records.push({ record_date: todayStr, normalized_value: state.last_known_rate }); } saveRecords(state.daily_records); renderUI(fetchTime); } } catch (err) { simulateFailure('offline'); } }
